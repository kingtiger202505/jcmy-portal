# 微信客服机器人依赖
#   python -m pip install -r requirements.txt
#   （start.bat 会自动带清华源 -i https://pypi.tuna.tsinghua.edu.cn/simple，
#     失败会退回默认 PyPI；手工装的话国内也建议加这个源）
#
# **这批包只在 Python 3.10~3.12 上有现成的安装包。**
# 用 3.13/3.14 装会报 "No matching distribution found for onnxruntime..."
# ——看着像网络问题，其实是 Python 版本太新，onnxruntime/opencv 还没出对应的包。
#
# 这里写的是**版本区间**而不是钉死的版本：本机（Python 3.11）验证过的确切版本写在下面，
# 但钉太死会在别的 Python 版本上直接装不上（实测：钉 pywin32==311，客户那台新 Python
# 只能装 312，整份 requirements 就卡死了）。
#   本机验证过的组合：
#     uiautomation 2.0.29 / pywin32 311 / rapidocr-onnxruntime 1.4.4 / pyperclip 1.11.0
#     requests 2.32.5 / numpy 2.2.6 / pillow 12.2.0 / onnxruntime 1.23.2
#     opencv-python 4.12.0.88 / shapely 2.1.1 / pyclipper 1.3.0.post6
#
# 这套代码是 **Windows 专用**（靠 win32 API 读窗口、发按键），Linux/macOS 跑不了。
#
# uiautomation          找微信窗口、把窗口抬到前台
# pywin32               截窗口图（BitBlt）、发按键、前台窗口管理
# rapidocr-onnxruntime  截图 OCR（群名、消息正文、输入框验证）—— 模型随包自带，不用另外下
# pyperclip             粘贴文字（避开输入法，比模拟打字可靠）
# requests              调 Dify 检索、企微表格 webhook
# numpy / pillow        OCR 和截图的底座
uiautomation>=2.0.20
pywin32>=306
rapidocr-onnxruntime>=1.4.4,<2
pyperclip>=1.8
requests>=2.31
numpy>=1.26,<3
pillow>=10

# OCR 链路的传递依赖。给区间而不是钉死，同样是为了在新 Python 上能装上；
# 上限挡住大版本 —— 群名、聊天标题、输入框内容全靠 OCR 认，
# 而"认的是哪个群"是按相似度卡阈值的（见 wechat_group_bot._name_matches），
# 识别结果漂一点点就可能认不出群，甚至认到隔壁名字相近的群上去。
onnxruntime>=1.18,<2
opencv-python>=4.9,<5
shapely>=2.0,<3
pyclipper>=1.3
